\section{The problem}

\begin{frame}{The problem}
  \[
    \zeta(3)=\sum_{m\ge1}\frac1{m^3}=1.2020569\ldots
    \qquad\text{Is it rational?}
  \]
  \begin{itemize}
    \item Ap\'ery (1978) proved that it is not. Beukers (1979) did it with integrals, Zudilin (2002) with sums.
    \item Today we go through Beukers' proof, see how it leads to Zudilin's, and then give a simpler proof of our own.
    \item Neither the series nor the decimal expansion tells us anything.
    \item We need an \alert{obstruction} to rationality.
  \end{itemize}
\end{frame}

\begin{frame}{The obstruction: no integer lies strictly between 0 and 1}
  If $\zeta(3)=p/q$ and $A,B\in\Z$, then $q(A\zeta(3)-B)\in\Z$.
  \medskip

  So it is enough to find numbers $J_n$ with
  \begin{enumerate}
    \item $\Dn^3J_n\in\Z+\Z\zeta(3)$, \quad $\Dn=\lcm(1,\dots,n)$ \hfill(arithmetic)
    \item $J_n>0$ \hfill(nonzero)
    \item $\Dn^3J_n\to0$ along some sequence of $n$ \hfill(small)
  \end{enumerate}
  \medskip
  The partial sums are not good enough: $\Dn^3\bigl(\zeta(3)-\sum_{m\le n}m^{-3}\bigr)\ge\frac{n^3}{2(n+1)^2}\to\infty$.
\end{frame}

\section{Part I: Beukers' integrals}

\begin{frame}{Step 1: writing $\zeta(3)$ as an integral}
  Integration turns powers into reciprocals of integers:
  \[
    \int_0^1x^m\dd x=\frac1{m+1},\qquad
    \int_0^1-x^m\log x\dd x=\frac1{(m+1)^2}\quad\text{(integrate by parts).}
  \]
  \pause
  Multiply the two, using a second variable $y$:
  \[
    \int_0^1\!\!\int_0^1-(xy)^m\log x\dd x\dd y=\frac1{(m+1)^3}.
  \]
  \pause
  Symmetrise: $-\log x-\log y=-\log(xy)$, so
  $\displaystyle\iint-(xy)^m\log(xy)\dd x\dd y=\frac2{(m+1)^3}$.
  \pause

  Now sum over $m\ge0$, using $\sum_m(xy)^m=\frac1{1-xy}$:
  \[
    \iint_{[0,1]^2}\frac{-\log(xy)}{1-xy}\dd x\dd y=2\zeta(3).
  \]
\end{frame}

\begin{frame}{Step 1 continued: playing with the integral}
  Call $K(x,y)=\frac{-\log(xy)}{1-xy}$ the \alert{kernel}. What happens if we multiply it by a monomial?
  \pause
  \begin{itemize}
    \item With $x^ry^r$ the sum starts later:
      \[
        \iint x^ry^rK=2\sum_{m>r}\frac1{m^3}=2\zeta(3)-2\Bigl(1+\frac1{2^3}+\dots+\frac1{r^3}\Bigr).
      \]
    \item With $x^ry^s$, $r\ne s$, the sum telescopes and $\zeta(3)$ drops out:
      \[
        \iint x^ry^sK=\frac1{r-s}\sum_{j=s+1}^{r}\frac1{j^2}\in\mathbb Q .
      \]
  \end{itemize}
  \pause
  \begin{block}{So every polynomial gives a linear combination of $\zeta(3)$ and $1$}
    For an integer polynomial $P$ of degree $\le n$:\quad
    $\Dn^3\iint P(x)P(y)K\in\Z+\Z\zeta(3)$.
    \\[2pt]
    For example, $P=1-2x$ gives $10\zeta(3)-12\approx0.0206$, which is already small. Can we do better?
  \end{block}
\end{frame}

\begin{frame}{Step 2: making the integral small, one variable first}
  We want an integer polynomial $P$ of degree $n$ that makes
  $\displaystyle\iint P(x)P(y)K(x,y)\dd x\dd y$ \alert{small}.
  \pause
  \medskip

  Start with one variable: how do we make $\int_0^1P(x)f(x)\dd x$ small? Expand $f$ in a Taylor series,
  \[
    f(x)=a_0+a_1x+\dots+a_{n-1}x^{n-1}+(\text{higher order terms}).
  \]
  \pause
  and ask $P$ to kill the low-degree monomials:
  \[
    \int_0^1x^jP(x)\dd x=0\qquad\text{for } j=0,1,\dots,n-1.
  \]
  Then $\int Pf$ only sees the higher-order part of $f$.
  \pause
  \begin{block}{This determines $P$ up to scale}
    There are $n$ conditions on $n+1$ coefficients, so the solution is unique up to scale.
  \end{block}
\end{frame}

\begin{frame}{Step 2 continued: the polynomial that kills monomials}
  The $n$-th derivative kills every polynomial of degree $<n$. Move it onto $P$ by integrating by
  parts $n$ times, with no boundary terms:
  \[
    P_n(x)=\frac1{n!}\frac{\dd^n}{\dd x^n}\bigl[x^n(1-x)^n\bigr],\qquad
    \int_0^1P_nf=\frac{(-1)^n}{n!}\int_0^1x^n(1-x)^nf^{(n)} .
  \]
  \pause
  $P_n=\sum_k(-1)^k\binom nk\binom{n+k}kx^k$ has integer coefficients (shifted Legendre polynomial).
  Take $P=P_n$:\quad
  $J_n=\iint P_n(x)P_n(y)K\dd x\dd y=2a_n\zeta(3)-2b_n$.
  \medskip

  \pause
  For example, take $f(x)=\frac1{x+t}$. Since
  $\frac{\dd^n}{\dd x^n}\frac1{x+t}=\frac{(-1)^nn!}{(x+t)^{n+1}}$,
  \[
    \int_0^1\frac{P_n(x)}{x+t}\dd x=\int_0^1\frac{[x(1-x)]^n}{(x+t)^{n+1}}\dd x\;>\;0 .
  \]
\end{frame}

\begin{frame}{Step 3: separation of variables}
  But $K(x,y)$ mixes $x$ and $y$, so how do we use the one-variable trick?
  \pause
  \medskip

  Borrow an idea from differential equations: write a function of two variables as a
  superposition of products, $u(x,t)=\sum_kv_k(x)w_k(t)$.
  \pause

  Here we write $K$ as an integral, over a parameter $t$, of products $v_t(x)\,w_t(y)$:
  \[
    K(x,y)=\int_0^\infty\underbrace{\frac1{x+t}}_{v_t(x)}\cdot\underbrace{\frac1{1+yt}}_{w_t(y)}\dd t .
  \]
  {\footnotesize (Partial fractions: $\int_0^\infty\frac{\dd t}{(t+a)(t+b)}=\frac{\log b-\log a}{b-a}$ with $a=x$, $b=1/y$, after writing $1+yt=y(t+b)$.)}
  \pause
  \[
    J_n=\int_0^\infty\Bigl(\int_0^1\frac{P_n(x)}{x+t}\dd x\Bigr)\Bigl(\int_0^1\frac{P_n(y)}{1+yt}\dd y\Bigr)\dd t
  \]
\end{frame}

\begin{frame}{Step 3 continued: positivity}
  Integrate by parts $n$ times in $x$ and in $y$; signs and factorials cancel:
  \[
    J_n=\int_0^\infty\!\!\iint_{[0,1]^2}
    \frac{[x(1-x)\,y(1-y)\,t]^n}{[(x+t)(1+yt)]^{n+1}}\dd x\dd y\dd t\;>\;0.
  \]
  The cancellation inside $P_n$ has turned into a positive integrand, and we have not estimated anything yet.
\end{frame}

\begin{frame}{Step 4: exponential decay}
  $J_n=\displaystyle\int\frac{G^n}{Q}$ \ with \ $Q=(x+t)(1+yt)$ \ and \ $G=\dfrac{x(1-x)\,y(1-y)\,t}{Q}$.
  \medskip

  If $G\le\rho$ everywhere, then $J_n\le\rho^n\displaystyle\int\frac1Q=2\zeta(3)\rho^n$.
  \pause
  \medskip

  AM--GM, used twice, reduces three variables to one. Put $u=\sqrt{xy}$.
  \begin{itemize}
    \item Denominator: $\dfrac Qt=1+xy+\underbrace{yt+\dfrac xt}_{\ge2\sqrt{xy}}\ge(1+u)^2$
      \newline\mbox{}\hfill{\small equality when $t=\sqrt{x/y}$ balances $yt$ and $x/t$}
    \item Numerator: $(1-x)(1-y)=1-\underbrace{(x+y)}_{\ge2\sqrt{xy}}+xy\le(1-u)^2$
      \newline\mbox{}\hfill{\small for fixed $xy$, equality when $x=y$}
  \end{itemize}
\end{frame}

\begin{frame}{Step 4 continued: a one-variable problem}
  \[
    G=\frac{xy\,(1-x)(1-y)}{Q/t}\le\Bigl(\underbrace{\frac{u(1-u)}{1+u}}_{g(u)}\Bigr)^2 .
  \]
  \pause
  Maximise $g$ by calculus. The numerator of $g'(u)$ is
  \[
    (1-2u)(1+u)-u(1-u)=1-2u-u^2,
  \]
  which vanishes at $u=\alpha=\sqrt2-1$.
  \pause
  \medskip

  Using $1-\alpha=\sqrt2\,\alpha$ and $1+\alpha=\sqrt2$:\quad $g(\alpha)=\alpha^2$.
\end{frame}

\begin{frame}{Step 4 continued: why $(u-\alpha)^2$?}
  We can also check the maximum without calculus. We want $\alpha^2(1+u)-u(1-u)\ge0$.
  \medskip

  This quadratic in $u$ would then be $\ge0$ and vanish at $u=\alpha$, so $\alpha$ would be a
  double root, and the quadratic a perfect square. Indeed:
  \[
    \alpha^2(1+u)-u(1-u)=u^2-(1-\alpha^2)u+\alpha^2=(u-\alpha)^2,
  \]
  and this is true because $1-\alpha^2=2\alpha$, i.e.\ $\alpha^2+2\alpha-1=0$.
  \pause
  \begin{block}{Result}
    $G\le\alpha^4=(\sqrt2-1)^4=17-12\sqrt2\approx\frac1{34}$, \ equality at $x=y=\alpha$, $t=1$:
    \[
      0<J_n\le2\zeta(3)\,(17-12\sqrt2)^n,\qquad\lim J_n^{1/n}=17-12\sqrt2.
    \]
  \end{block}
\end{frame}

\begin{frame}{Step 5: least common multiples, without the prime number theorem}
  \begin{block}{Identity}
    \[
      N!=\prod_{k=1}^ND_{\lfloor N/k\rfloor}
    \]
    (compare the exponent of each prime $p$ on both sides).
  \end{block}
  \begin{itemize}
    \item If $D_m\ge\mu^m$ for all large $m$, then $\log N!\gtrsim\log\mu\cdot N\log N$.
    \item So for every $\mu>e$ we have $\Dn<\mu^n$ for \alert{infinitely many} $n$.
    \item Take $\mu=3$. This is all we need, because $J_n>0$ for every $n$.
  \end{itemize}
\end{frame}

\section{Part I: the contradiction}

\begin{frame}{The contradiction}
  Suppose $\zeta(3)=p/q$. Then $q\Dn^3J_n$ is a \alert{positive integer}, and for infinitely many $n$
  \[
    1\le q\Dn^3J_n<2q\zeta(3)\bigl[27(17-12\sqrt2)\bigr]^n\longrightarrow0,
  \]
  since $17+12\sqrt2>32$ gives $27(17-12\sqrt2)<\tfrac{27}{32}<1$.
  \begin{alertblock}{Theorem (Ap\'ery)}
    $\zeta(3)$ is irrational.
  \end{alertblock}
\end{frame}

\begin{frame}{The first few values}
  \centering
  \begin{tabular}{@{}rrll@{}}
    \toprule
    $n$ & $a_n$ & $J_n=2a_n\zeta(3)-2b_n$ & value\\
    \midrule
    1 & 5 & $10\zeta(3)-12$ & $2.06\times10^{-2}$\\
    2 & 73 & $146\zeta(3)-\tfrac{351}{2}$ & $3.08\times10^{-4}$\\
    3 & 1445 & $2890\zeta(3)-\tfrac{62531}{18}$ & $5.69\times10^{-6}$\\
    \bottomrule
  \end{tabular}
  \medskip

  Both terms are large and their difference is tiny, but by Step 3 it is never zero.
\end{frame}

\section{Part II: from integrals to sums}

\begin{frame}{Zudilin's rational function is our Legendre polynomial}
  Collect all the moments of $P_n$ in one function:
  \[
    r_n(t)=\int_0^1x^{t-1}P_n(x)\dd x\qquad(t>0).
  \]
  \pause
  Integrate by parts $n$ times, as in Step 2:
  \[
    \boxed{\;r_n(t)=\frac{(1-t)(2-t)\cdots(n-t)}{t\,(t+1)\cdots(t+n)}\;}
  \]
  \pause
  We can see where the zeros and the poles come from:
  \begin{itemize}
    \item the zeros at $t=1,\dots,n$ are the vanishing moments $\int_0^1x^jP_n=0$ we asked for;
    \item the poles at $t=0,-1,\dots,-n$ come from integrating the monomials $x^k$, $k\le n$.
  \end{itemize}
\end{frame}

\begin{frame}{Why a square?}
  Expand the kernel's denominator, indexing the terms by $t=1,2,3,\dots$:
  \[
    \frac1{1-xy}=1+xy+(xy)^2+(xy)^3+\cdots=\sum_{t=1}^\infty(xy)^{t-1}.
  \]
  \pause
  Then Beukers' integral becomes a sum of integrals:
  \[
    J_n=\iint P_n(x)P_n(y)\,\frac{-\log(xy)}{1-xy}
    =\sum_{t=1}^\infty\iint P_n(x)P_n(y)\,(xy)^{t-1}\bigl(-\log(xy)\bigr).
  \]
  \pause
  Ignore the logarithm for a moment. Since $(xy)^{t-1}=x^{t-1}\,y^{t-1}$, with the same
  exponent on $x$ and $y$, each integral splits into two equal factors:
  \[
    \iint P_n(x)P_n(y)\,x^{t-1}y^{t-1}
    =\Bigl(\int_0^1x^{t-1}P_n\Bigr)\Bigl(\int_0^1y^{t-1}P_n\Bigr)=r_n(t)^2=:R_n(t).
  \]
\end{frame}

\begin{frame}{The derivative comes from the logarithm}
  Now put the logarithm back. It appears when we differentiate in the exponent:
  \[
    \frac{\dd}{\dd t}(xy)^{t-1}=\log(xy)\,(xy)^{t-1}.
  \]
  \pause
  So the $t$-th term of the sum is
  \[
    \iint P_n(x)P_n(y)\,(xy)^{t-1}\bigl(-\log(xy)\bigr)
    =-\frac{\dd}{\dd t}\iint P_n(x)P_n(y)\,(xy)^{t-1}=-R_n'(t).
  \]
  \pause
  Adding up all the terms:
  \[
    \boxed{\;J_n=-\sum_{t=1}^\infty R_n'(t)\;}
  \]
\end{frame}

\begin{frame}{The same recipe, without integrals}
  \[
    J_n=-\sum_{t=1}^\infty R_n'(t),\qquad R_n(t)=\Bigl(\frac{(1-t)(2-t)\cdots(n-t)}{t\,(t+1)\cdots(t+n)}\Bigr)^2
  \]
  \begin{itemize}
    \item The poles at $0,-1,\dots,-n$ give shifted zeta values with controlled denominators.
    \item The zeros at $1,\dots,n$ make the first $n$ terms of the sum vanish.
    \item Because of the square the zeros are double, so they survive the derivative.
    \item With the derivative every pole gains a power, which gives cubes and hence $\zeta(3)$.
  \end{itemize}
\end{frame}

\begin{frame}{The arithmetic, from partial fractions}
  \[
    r_n(t)=\sum_{k=0}^n\frac{c_k}{t+k},\qquad c_k=(-1)^k\frac{(n+k)!}{k!^2\,(n-k)!}\in\Z .
  \]
  \pause
  \begin{itemize}
    \item Squares: $\;-\sum_t\frac{\dd}{\dd t}\frac1{(t+k)^2}=2\Bigl(\zeta(3)-1-\frac1{2^3}-\dots-\frac1{k^3}\Bigr)$.
    \item Mixed terms: $\frac1{(t+k)(t+j)}=\frac1{j-k}\bigl(\frac1{t+k}-\frac1{t+j}\bigr)$, so $\zeta(2)$
      cancels.
  \end{itemize}
  \pause
  \[
    J_n=2a_n\zeta(3)-2b_n,\qquad \Dn^3b_n\in\Z,\qquad a_n=\sum_kc_k^2 .
  \]
  The $a_n=1,5,73,1445,\dots$ are Ap\'ery's numbers, and $J_1=10\zeta(3)-12$ as before.
\end{frame}

\begin{frame}{The catch: the terms have both signs}
  The first $n$ terms vanish, so $J_n=-\sum_{t>n}R_n'(t)$. The easy bound
  $|R_n'(t)|\le\frac{n+2}{t^3}$ gives
  \[
    |J_n|\le(n+2)\sum_{t>n}\frac1{t^3}<\frac{n+2}{2n(n+1)} .
  \]
  \pause
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The sum gives us}
      \begin{itemize}
        \item $J_n\to0$, but only like $\frac1{2n}$;
        \item no information on the sign.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{We need}
      \begin{itemize}
        \item $J_n>0$;
        \item $J_n$ smaller than $\Dn^{-3}\approx20^{-n}$.
      \end{itemize}
    \end{column}
  \end{columns}
  \pause
  \medskip

  For Beukers the positive integral gives both. What can we use without it?
\end{frame}

\section{Part II: Zudilin's route}

\begin{frame}{Zudilin's plan: a second series for the same number}
  \centering
  \begin{tikzpicture}[box/.style={draw,rounded corners,align=center,inner sep=6pt,font=\small},
                      >=stealth]
    \node[box] (F) at (-4,0) {$J_n=-\sum R_n'(t)$\\[2pt] arithmetic \alert{\checkmark}\\ size \alert{?}};
    \node[box] (G) at (4,0) {Ball's series $\widetilde F_n=\sum\widetilde R_n(t)$\\[2pt]
      size \alert{\checkmark} (positive terms)\\ arithmetic \alert{?} (needs $\Dn^4$)};
    \node[box] (L) at (0,-2.3) {same recurrence\\ $(n+1)^3u_{n+1}-(2n+1)(17n^2+17n+5)u_n+n^3u_{n-1}=0$\\
      $+$ same values at $n=0,1$};
    \draw[->,thick] (F) -- (L);
    \draw[->,thick] (G) -- (L);
    \node[font=\small] at (0,0.3) {$J_n=\widetilde F_n$ ?};
  \end{tikzpicture}
  \medskip

  Each series has one of the two properties. A common recurrence shows that they are equal.
\end{frame}

\begin{frame}{Ball's series}
  \[
    \widetilde R_n(t)=n!^2\,(2t+n)\,
    \frac{(t-1)\cdots(t-n)\cdot(t+n+1)\cdots(t+2n)}{\bigl(t(t+1)\cdots(t+n)\bigr)^4}
  \]
  \begin{itemize}
    \item It is zero for $t\le n$ and positive for $t>n$, and a Stirling-type estimate gives
      $\widetilde F_n<20(n+1)^4(\sqrt2-1)^{4n}$.
    \item There is no obvious reason why it should equal $J_n$.
  \end{itemize}
  \pause
  For $n=1$ we can check by hand that it does:
  \[
    \widetilde R_1(t)=-\frac2{t^4}+\frac2{(t+1)^4}+\frac5{t^3}+\frac5{(t+1)^3}-\frac5{t^2}+\frac5{(t+1)^2}
  \]
  \[
    \widetilde F_1=-2\zeta(4)+2(\zeta(4)-1)+5\zeta(3)+5(\zeta(3)-1)-5\zeta(2)+5(\zeta(2)-1)=10\zeta(3)-12 .
  \]
\end{frame}

\begin{frame}{The certificate for Ball's series}
  The recurrence for $\widetilde F_n$ comes from a telescoping identity whose certificate is
  $\dfrac{\widetilde R_n(t)}{(2t+n)(t+2n-1)(t+2n)}$ times
  {\small
  \begin{align*}
    &-t^6-(8n-1)t^5+(4n^2+27n+5)t^4+2n(67n^2+71n+15)t^3\\
    &+(358n^4+339n^3+76n^2-7n-3)t^2\\
    &+(384n^5+396n^4+97n^3-29n^2-17n-2)t\\
    &+n(153n^5+183n^4+50n^3-30n^2-22n-4).
  \end{align*}}
  \pause
  To check it, expand a polynomial identity in $t$ and $n$. To find it, use a computer.
\end{frame}

\begin{frame}{The end of Zudilin's proof}
  Zudilin finishes ``using a trivial bound $\Dn<3^n$''. Our Step 5 is enough for this:
  we only need $\Dn<3^n$ for infinitely many $n$, because $\widetilde F_n>0$ for every $n$:
  \[
    0<q\,\Dn^3J_n<20q\,(n+1)^4\bigl[27(\sqrt2-1)^4\bigr]^n\longrightarrow0
    \quad\text{along those }n .
  \]
  \pause
  \begin{block}{Zudilin's proof in total}
    one series for arithmetic $+$ one series for size $+$ two recurrences
    $+$ two certificates $+$ a bridge $+$ Stirling-type estimates.
  \end{block}
\end{frame}

\section{Part III: our proof, a maximum principle}

\begin{frame}{What we have, and what we are missing}
  We keep only the first series, $J_n=-\sum_tR_n'(t)$. We already know:
  \begin{itemize}
    \item arithmetic: $J_n=2a_n\zeta(3)-2b_n$, \ $\Dn^3b_n\in\Z$;
    \item $J_0=2\zeta(3)>0$;
    \item $J_n\to0$.
  \end{itemize}
  \pause
  \medskip
  We are still missing the \alert{sign} of $J_n$ and its \alert{rate of decay}.
  \pause
  \medskip

  Instead of a second series, we look for an equation relating each $J_n$ to its
  neighbours, and let that equation control the sequence.
\end{frame}

\begin{frame}{Which equation would we like?}
  Suppose \qquad $B\,(J_{n+1}-J_n)-A\,(J_n-J_{n-1})=\lambda J_n$, \qquad $A,B,\lambda>0$.
  \pause
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.8]
        \draw[gray] (-0.3,0)--(3.3,0);
        \foreach \x/\y in {0/-0.4,1.5/-1.2,3/-0.6} \fill (\x,\y) circle (2.5pt);
        \draw (0,-0.4)--(1.5,-1.2)--(3,-0.6);
        \node[below,font=\small] at (1.5,-1.2) {$J_n<0$};
        \node[above,font=\small] at (1.5,0) {negative valley};
      \end{tikzpicture}
      \\ left side $\ge0$, right side $<0$
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.8]
        \draw[gray] (-0.3,0)--(3.3,0);
        \foreach \x/\y in {0/0.6,1.5/1.2,3/0.4} \fill (\x,\y) circle (2.5pt);
        \draw (0,0.6)--(1.5,1.2)--(3,0.4);
        \node[above,font=\small] at (1.5,1.2) {$J_n>0$};
        \node[below,font=\small] at (1.5,0) {positive peak};
      \end{tikzpicture}
      \\ left side $\le0$, right side $>0$
    \end{column}
  \end{columns}
  \pause
  \medskip
  Neither can happen. And once $J_n$ is positive and decreasing,
  \[
    A(J_{n-1}-J_n)>\lambda J_n\quad\Longrightarrow\quad J_n<\frac{A}{A+\lambda}\,J_{n-1},
  \]
  so a large $\lambda/A$ means fast decay. \ Is there such an equation?
\end{frame}

\begin{frame}{A guess}
  Look for it among the rational functions $R_j=r_j^2$ themselves:
  \[
    B(R_{n+1}-R_n)-A(R_n-R_{n-1})=\lambda R_n+\underbrace{W(t+1)R_n(t+1)-W(t)R_n(t)}_{\text{a telescoping error}}
  \]
  \pause
  \begin{itemize}
    \item We take consecutive indices because $r_{n\pm1}$ differs from $r_n$ by a single factor.
    \item The error telescopes, so after $-\sum_t\frac{\dd}{\dd t}$ it leaves only boundary terms,
      which vanish (double zero of $R_n$ at $t=1$; finite limit at $\infty$).
    \item We take $W$ a polynomial of degree $\le2$: no new poles, and $R_n\sim t^{-2}$ keeps
      $WR_n$ bounded.
  \end{itemize}
  \pause
  That leaves six unknowns: $A$, $B$, $\lambda$ and the coefficients of $W(t)=w_0+w_1t+w_2t^2$.
\end{frame}

\begin{frame}{Six unknowns, five equations}
  Divide by $R_n$. All the ratios are simple:
  \[
    \frac{R_{n-1}}{R_n}=\Bigl(\frac{t+n}{t-n}\Bigr)^2,\quad
    \frac{R_{n+1}}{R_n}=\Bigl(\frac{t-n-1}{t+n+1}\Bigr)^2,\quad
    \frac{R_n(t+1)}{R_n(t)}=\frac{t^4}{(n-t)^2(n+1+t)^2}.
  \]
  \pause
  The identity becomes
  \[
    \frac{4Ant}{(n-t)^2}-\frac{4B(n+1)t}{(n+1+t)^2}=\lambda-W(t)+\frac{t^4\,W(t+1)}{(n-t)^2(n+1+t)^2}.\tag{$\star$}
  \]
  \pause
  Multiplied by $(n-t)^2(n+1+t)^2$, both sides have degree $\le4$ (the $t^6$ and $t^5$ terms
  cancel). That makes five equations for six unknowns, so a solution exists.
  \medskip

  That factor is nonzero at $t=0$, so it is enough to match Taylor coefficients at $t=0$ up to $t^4$.
\end{frame}

\begin{frame}{Solving it, one coefficient at a time}
  Expand both sides of $(\star)$ at $t=0$:
  \medskip

  \centering
  \begin{tabular}{@{}ll@{}}
    \toprule
    power & matching the coefficients gives\\
    \midrule
    $t^3$ & $12\bigl(\frac A{n^3}-\frac B{(n+1)^3}\bigr)=0$, \ so \ $A=n^3$, \ $B=(n+1)^3$\\[4pt]
    \pause
    $t^0,t^1,t^2$ & $w_0=\lambda$, \ $w_1=4(2n+1)$, \ $w_2=-8(2n+1)$\\[4pt]
    \pause
    $t^4$ & $\dfrac{16(2n+1)}{n(n+1)}=\dfrac{W(1)}{n^2(n+1)^2}=\dfrac{\lambda-4(2n+1)}{n^2(n+1)^2}$,
      \ so \ $\lambda=4(2n+1)^3$\\
    \bottomrule
  \end{tabular}
  \pause
  \medskip
  \begin{block}{The solution}
    \centering
    $A=n^3,\qquad B=(n+1)^3,\qquad \lambda=4(2n+1)^3$, \qquad all positive.
  \end{block}
\end{frame}

\begin{frame}{The current law}
  Apply $-\sum_t\frac{\dd}{\dd t}$. The telescoping error disappears, and we get
  \[
    (n+1)^3(J_{n+1}-J_n)-n^3(J_n-J_{n-1})=4(2n+1)^3J_n .
  \]
  \pause
  \centering
  \begin{tikzpicture}[x=3.2cm,>=stealth]
    \draw[thick] (0,0)--(2,0);
    \foreach \i/\lab in {0/{J_{n-1}},1/{J_n},2/{J_{n+1}}} {
      \fill (\i,0) circle (2.5pt); \node[above=3pt] at (\i,0) {$\lab$}; }
    \node[above=1pt,font=\small] at (0.5,0) {$n^3$};
    \node[above=1pt,font=\small] at (1.5,0) {$(n+1)^3$};
    \draw[->,thick,blue] (0.3,-0.3)--(0.7,-0.3) node[midway,below,font=\small] {$C_n$};
    \draw[->,thick,blue] (1.4,-0.3)--(1.8,-0.3) node[midway,below,font=\small] {$C_{n+1}$};
    \draw[thick] (1,0)--(1,-1.3);
    \draw[thick] (0.94,-1.3)--(1.06,-1.3) (0.96,-1.37)--(1.04,-1.37) (0.985,-1.44)--(1.015,-1.44);
    \draw[->,thick,blue] (1.04,-0.45)--(1.04,-0.95);
    \node[right,font=\small] at (1.06,-0.7) {$4(2n+1)^3J_n$};
  \end{tikzpicture}
  \raggedright
  \pause

  Read $J_n$ as a voltage, with wires of conductance $n^3$, a leak of conductance $4(2n+1)^3$
  and current $C_n=n^3(J_{n-1}-J_n)$. \ Kirchhoff's law at node $n$ says
  \[
    C_n=C_{n+1}+4(2n+1)^3J_n,\qquad\text{what flows in either flows on or leaks away.}
  \]
\end{frame}

\begin{frame}{Where does the 17 come from?}
  Our current law, \ $(n+1)^3(J_{n+1}-J_n)-n^3(J_n-J_{n-1})=4(2n+1)^3J_n$, \ grouped by $J_{n+1},J_n,J_{n-1}$:
  \[
    (n+1)^3J_{n+1}-\bigl[\underbrace{(n+1)^3+n^3}_{\text{two wires}}+\underbrace{4(2n+1)^3}_{\text{leak}}\bigr]J_n+n^3J_{n-1}=0 .
  \]
  Both parts of the middle coefficient contain the factor $2n+1$:
  \[
    \begin{aligned}
      (n+1)^3+n^3&=(2n+1)\,(n^2+n+1)\\
      4(2n+1)^3&=(2n+1)\,(16n^2+16n+4)\\
      \text{sum}&=(2n+1)\,(17n^2+17n+5)
    \end{aligned}
  \]
  which is Ap\'ery's recurrence:
  \ $(n+1)^3J_{n+1}-(2n+1)(17n^2+17n+5)J_n+n^3J_{n-1}=0$.
  \begin{block}{}
    \centering $17=1+16$: \ one from the two wires, sixteen from the leak.
  \end{block}
\end{frame}

\begin{frame}{No valleys, no peaks: $J_n$ decreases to zero}
  A sequence tending to $0$ attains its minimum if it has a negative term, and attains its
  maximum if its terms are positive.
  \pause
  \begin{enumerate}
    \item[(a)] \textbf{Never negative.} Otherwise the minimum is negative, attained at some $n\ge1$
      (as $J_0>0$), which is a negative valley.
    \pause
    \item[(b)] \textbf{Never zero.} If $J_n=0$, then $(n+1)^3J_{n+1}+n^3J_{n-1}=0$ with both terms
      $\ge0$, so both vanish, and going down we reach $J_0=0$.
    \pause
    \item[(c)] \textbf{Decreasing.} If $J_n\ge J_{n-1}$, the largest of $J_n,J_{n+1},\dots$ is
      attained at some $k\ge n$, which is a positive peak.
  \end{enumerate}
  \pause
  \[
    0<J_n<J_{n-1}\qquad\text{for all }n\ge1 .
  \]
\end{frame}

\begin{frame}{From falling to falling fast}
  Since $J_n$ decreases, $C_{n+1}>0$, so the incoming current is larger than the leak:
  \[
    n^3(J_{n-1}-J_n)>4(2n+1)^3J_n\quad\Longrightarrow\quad
    \frac{J_{n-1}}{J_n}>1+4\Bigl(2+\frac1n\Bigr)^3>33 .
  \]
  \[
    \boxed{\;0<J_n\le2\zeta(3)\,33^{-n}\;}
  \]
  \pause
  \centering
  \begin{tabular}{@{}lcccccc@{}}
    \toprule
    $n$ & 1 & 2 & 3 & 4 & 5 & $\infty$\\
    \midrule
    $J_{n-1}/J_n$ & 116.9 & 66.8 & 54.1 & 48.5 & 45.3 & $17+12\sqrt2\approx33.97$\\
    \bottomrule
  \end{tabular}
  \medskip

  The maximum principle loses very little: 33 against the true 33.97.
\end{frame}

\begin{frame}{What if there were no leak?}
  Then the current is constant: $C_n=C$, so $J_{n-1}-J_n=\dfrac C{n^3}$, and since $J_n\to0$,
  \[
    J_n=C\sum_{k>n}\frac1{k^3}.
  \]
  \pause
  This is the tail of $\zeta(3)$, that is, the partial sums we rejected at the start, and it
  decays only like $\frac1{2n^2}$.
  \pause
  \begin{block}{}
    The approximation that fails and the one that works satisfy the same current law.\\
    The only difference is the leak, which is more than $32$ times the conductance at every node.
  \end{block}
\end{frame}

\begin{frame}{Denominators, without prime numbers}
  The same identity as in Part I: \quad $N!=\prod_{k=1}^ND_{\lfloor N/k\rfloor}$.
  \pause
  \begin{block}{Uniform version}
    If $D_k\ge\kappa\,\mu^k$ for \emph{all} $k\ge1$ (with $\kappa>0$, $\mu\ge1$), then $\mu\le e$.
  \end{block}
  \pause
  \textbf{Proof.} Each index $\lfloor N/k\rfloor\ge1$ and $\sum_k\lfloor N/k\rfloor>N(\log N-1)$, so
  \[
    N^N\ge N!\ge\kappa^N\mu^{N(\log N-1)}\quad\Longrightarrow\quad N\ge\frac\kappa\mu\,N^{\log\mu}\ \text{ for all }N
    \quad\Longrightarrow\quad\log\mu\le1 .
  \]
\end{frame}

\begin{frame}{The contradiction again}
  Suppose $\zeta(3)=p/q$. Then $\frac q2\Dn^3J_n=\Dn^3(pa_n-qb_n)$ is a \alert{positive integer}, so
  \[
    \Dn^3\ge\frac2{qJ_n}\ge\frac{33^n}{q\,\zeta(3)}=\frac{33^n}{p}\qquad\text{for every }n.
  \]
  \pause
  So $\Dn\ge p^{-1/3}\bigl(33^{1/3}\bigr)^n$ for all $n$, hence $33^{1/3}\le e$, i.e.\ $33\le e^3$.
  \pause
  But $e<3$, so $e^3<27<33$.
  \begin{alertblock}{Theorem (Ap\'ery)}
    $\zeta(3)$ is irrational.
  \end{alertblock}
\end{frame}

  \centering
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
     & Zudilin & Ours\\

\end{frame}
